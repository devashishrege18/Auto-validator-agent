# -------------------------------------------------------
# requirements.txt
# -------------------------------------------------------
# These are the Python packages needed to run the
# Auto-Auditor Validator Agent backend.
#
# Install them with:
#   pip install -r requirements.txt
# -------------------------------------------------------

fastapi           # Web framework for building APIs
uvicorn[standard] # ASGI server to run the FastAPI app
requests          # HTTP client for the validator agent to call compliance APIs
scikit-learn      # ML framework for training the compliance model
pandas            # Data manipulation for the training pipeline
joblib            # Model serialisation (save/load .pkl files)
pytest            # Unit testing framework
httpx             # Async HTTP client (required by FastAPI TestClient)
